function Callout({
  x1,
  y1,
  x2,
  label,
  align = "right",
}: {
  x1: number;
  y1: number;
  x2: number;
  label: string;
  align?: "left" | "right";
}) {
  const textX = align === "right" ? x2 + 6 : x2 - 6;
  return (
    <g>
      <path
        d={`M${x1} ${y1} L${x2} ${y1}`}
        stroke="currentColor"
        strokeWidth="0.75"
      />
      <circle cx={x1} cy={y1} r="1.6" fill="currentColor" />
      <text
        x={textX}
        y={y1 + 3}
        fill="currentColor"
        fontSize="7.5"
        letterSpacing="0.6"
        fontFamily="var(--font-mono)"
        textAnchor={align === "right" ? "start" : "end"}
      >
        {label}
      </text>
    </g>
  );
}
