"use client";

import { useEffect, useRef, useState } from "react";

const STROKE = "currentColor";

function Frame({ label }: { label: string }) {
  return (
    <div className="relative border border-hairline bg-surface">
      <svg
        viewBox="0 0 100 100"
        className="pointer-events-none absolute inset-0 h-full w-full text-ink"
        aria-hidden="true"
      >
        {[
          [4, 4],
          [96, 4],
          [4, 96],
          [96, 96],
        ].map(([x, y]) => (
          <path
            key={`${x}-${y}`}
            d={`M${x - 3} ${y} h6 M${x} ${y - 3} v6`}
            stroke={STROKE}
            strokeWidth="0.6"
            opacity="0.35"
          />
        ))}
      </svg>
      <span className="meta absolute bottom-2.5 right-3">{label}</span>
    </div>
  );
}

export function LayerReveal() {
  const ref = useRef<HTMLDivElement>(null);
  const [progress, setProgress] = useState(0);
  const [reduced, setReduced] = useState(false);

  useEffect(() => {
    const node = ref.current;
    if (!node) return;

    const rm = window.matchMedia("(prefers-reduced-motion: reduce)").matches;
    if (rm) {
      setReduced(true);
      setProgress(1);
      return;
    }

    let frame = 0;
    const update = () => {
      frame = 0;
      const rect = node.getBoundingClientRect();
      const total = rect.height + window.innerHeight;
      const scrolled = window.innerHeight - rect.top;
      const next = Math.min(1, Math.max(0, scrolled / total));
      setProgress(next);
    };

    const onScroll = () => {
      if (!frame) frame = window.requestAnimationFrame(update);
    };

    update();
    window.addEventListener("scroll", onScroll, { passive: true });
    window.addEventListener("resize", onScroll);
    return () => {
      if (frame) window.cancelAnimationFrame(frame);
      window.removeEventListener("scroll", onScroll);
      window.removeEventListener("resize", onScroll);
    };
  }, []);

  const uniformOpacity = 1 - Math.min(1, progress * 1.5) * 0.72;
  const underOpacity = 0.35 + Math.min(1, progress * 3) * 0.55;
  const outlineOpacity = Math.max(0, progress * 1.6 - 0.35);
  const linerT = Math.min(1, Math.max(0, (progress - 0.25) / 0.75));
  const linerOpacity = Math.min(1, 0.35 + linerT * 1.5);
  const linerShift = linerT * 46;

  return (
    <div ref={ref}>
      <div className="relative aspect-[4/5] w-full">
        <Frame label="CV-A / LAYER STUDY" />

        <div
          className="absolute inset-0 grid place-items-center p-8"
          role="img"
          aria-label="Exploded layer study: the outer uniform shirt, the sleeveless undershirt worn beneath it, and the removable phase-change liner separating out below."
        >
          <div className="relative aspect-[3/4] w-full max-w-[280px]">
            {/* uniform shirt */}
            <svg
              viewBox="0 0 200 260"
              className="absolute inset-0 h-full w-full text-ink"
              style={{ opacity: uniformOpacity }}
              aria-hidden="true"
            >
              <path
                d="M62 34 h76 l30 20 -16 26 -8 -5 v140 a4 4 0 0 1 -4 4 H60 a4 4 0 0 1 -4 -4 V75 l-8 5 -16 -26 z"
                fill="#EFEDE8"
                stroke={STROKE}
                strokeWidth="1.25"
                strokeLinejoin="round"
              />
              <path
                d="M78 34 q22 16 44 0"
                fill="none"
                stroke={STROKE}
                strokeWidth="1.25"
              />
              <path
                d="M100 50 V76"
                stroke={STROKE}
                strokeWidth="1.25"
                opacity="0.4"
              />
            </svg>

            {/* undershirt, dashed = worn beneath */}
            <svg
              viewBox="0 0 200 260"
              className="absolute inset-0 h-full w-full text-ink"
              style={{ opacity: underOpacity }}
              aria-hidden="true"
            >
              <path
                d="M70 56 q30 -12 60 0 l14 16 v118 a4 4 0 0 1 -4 4 H60 a4 4 0 0 1 -4 -4 V72 z"
                fill="none"
                stroke={STROKE}
                strokeWidth="1.25"
                strokeDasharray="5 4"
              />
            </svg>

            {/* uniform reduced to outline */}
            <svg
              viewBox="0 0 200 260"
              className="absolute inset-0 h-full w-full text-navy"
              style={{ opacity: outlineOpacity * 0.55 }}
              aria-hidden="true"
            >
              <path
                d="M62 34 h76 l30 20 -16 26 -8 -5 v140 a4 4 0 0 1 -4 4 H60 a4 4 0 0 1 -4 -4 V75 l-8 5 -16 -26 z"
                fill="none"
                stroke={STROKE}
                strokeWidth="1"
              />
            </svg>

            {/* the liner, separating out */}
            <svg
              viewBox="0 0 200 260"
              className="absolute inset-0 h-full w-full text-accent"
              style={{
                opacity: linerOpacity,
                transform: `translateY(${linerShift}px)`,
                transition: "transform 120ms linear",
              }}
              aria-hidden="true"
            >
              <path
                d="M74 62 q26 -10 52 0 l10 14 v116 a4 4 0 0 1 -4 4 H68 a4 4 0 0 1 -4 -4 V76 z"
                fill="#C2603A"
                fillOpacity="0.05"
                stroke={STROKE}
                strokeWidth="1.25"
              />
            </svg>
          </div>
        </div>

        <p className="meta absolute bottom-2.5 left-3">
          {reduced ? "Layers shown separated" : "Scroll to separate the layers"}
        </p>
      </div>
      <p className="mt-5 max-w-[54ch] text-[14px] leading-relaxed text-ink-muted">
        Exploded layer study: the uniform already worn, the sleeveless
        undershirt beneath it, and the liner lifting out as one piece. It is a
        drawing of how the three are meant to stack — not a photograph, and not
        a measurement of anything.
      </p>
    </div>
  );
}
